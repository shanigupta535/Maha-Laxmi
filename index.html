Connect a Real Database (Firebase — free)

To sync products across devices, connect Firebase Firestore. Get a free project at console.firebase.google.com, then replace the storage functions in the code with these:

// 1) Add Firebase SDK in <head>
<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
import { getFirestore, collection, getDocs, addDoc,
         updateDoc, deleteDoc, doc }
  from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js";

const app = initializeApp({ /* your firebaseConfig */ });
const db = getFirestore(app);
const productsRef = collection(db, "products");

async function loadProductsFromFirebase(){
  const snap = await getDocs(productsRef);
  return snap.docs.map(d => ({ id: d.id, ...d.data() }));
}
async function addProductToFirebase(product){
  const ref = await addDoc(productsRef, product);
  return ref.id;
}
async function updateProductInFirebase(id, data){
  await updateDoc(doc(db, "products", id), data);
}
async function deleteProductFromFirebase(id){
  await deleteDoc(doc(db, "products", id));
}
</script>

Note: Firebase free tier is plenty for a small shop.
